#include<stdio.h>

struct Student
{
    int roll;
    int m1,m2,m3;
    int total;
    float per;
    char name[50];
};

int main()
{
    int i,n;

    printf("---------Student Information----------");
    printf("\n Enter the no. of records:");
    scanf("%d",&n);

    struct Student starr[n];

    for(i=0;i<n;i++)
    {
        printf("Enter the roll no. of student :");
        scanf("%d",&starr[i].roll);

        printf("Enter the marks of 3 subjects :");
        scanf("%d%d%d",&starr[i].m1,&starr[i].m2,&starr[i].m3);

        printf("Enter the name of the student");
        scanf("%s",&starr[i].name);

        starr[i].total = starr[i].m1 + starr[i].m2 + starr[i].m3;
        starr[i].per= starr[i].total/3.0;
    }

    printf("\n Roll no. |\t Name |\t\t\t Marks |\t\t Total |\t Percentage |\t Result");
     
    for(i=0;i<n;i++)
    {

        printf("\n%d",starr[i].roll);
        printf("\t\t");
        printf("%s",starr[i].name);
        printf("\t\t");
        printf("%d",starr[i].m1);
        printf("\t%d",starr[i].m2);
        printf("\t%d",starr[i].m3);
        printf("\t\t%d",starr[i].total);
        printf("\t\t%f\t",starr[i].per);

        if(starr[i].per<20)
        printf("Fail");
        else
        printf("Pass");
    }
    return 0;
}
