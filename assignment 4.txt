#include<stdio.h>
#include<stdlib.h>
void main()
{
    int a[2][3],b[2][3],sum[2][3],i,j;
    printf("Enter values on first matrix\n");
    for(i=0;i<2;i++)
    {
        for(j=0;j<3;j++)
        {
            scanf("%d",&a[i][j]);
            printf("\t");
        }
        printf("\n");
    }

    printf("Enter values in the second matrix\n");
    for(i=0;i<2;i++)
    {
        for(j=0;j<3;j++)
        {
            scanf("%d",&b[i][j]);
            printf("\t");
        }
        printf("\n");
    }

    /*Sum*/
    for(i=0;i<2;i++)
    {
        for(j=0;j<3;j++)
        {
            sum[i][j]=a[i][j]+b[i][j];
        }
    }
     printf("Sum of the matrices is:\n");
     for(i=0;i<2;i++)
     {
        for(j=0;j<3;j++)
        {
            printf("%d",sum[i][j]);
            printf("\t");
        }
        printf("\n");
     }

}
