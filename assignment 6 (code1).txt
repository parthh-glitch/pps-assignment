#include<stdio.h>
int fact();
int main()
{
    int num,result;
    printf("Enter a number for its factorial");
    scanf("%d",&num);
    result=fact(num);
    printf("The factorial of this number = %d",result);
    return 0;
}

int fact(int n)
{
    int f=1,i;
    for(i=1;i<=n;i++)
    {
        f=f*i;
    }
    return f;
}
